Задание

\[\textbf{Вычислить интеграл: } \int_{0}^{+\infty} \frac{x\ln x}{(1+x^{2})^{2}} \,dx. \]

Решение

\[ \textbf{Введём замену } t = \frac{1}{x}: \space dx = -\frac{1}{t^{2}} dt, \space \lim_{x \to 0^{+}} \frac{1}{x} = +\infty, \space \lim_{x \to +\infty} \frac{1}{x} = 0;\] \[ \int_{0}^{+\infty} \frac{x\ln x}{(1+x^{2})^{2}} \,dx = \int_{+\infty}^{0} \frac{\frac{1}{t}\ln\frac{1}{t}}{\left( 1+\frac{1}{t^{2}}\right)^{2}} \left( -\frac{1}{t^{2}}\right) \,dt = -\int_{0}^{+\infty} \frac{t\ln t}{(t^{2}+1)^{2}}\,dt \Rightarrow \int_{0}^{+\infty} \frac{x\ln x}{(1+x^{2})^{2}} \,dx = 0.\]
