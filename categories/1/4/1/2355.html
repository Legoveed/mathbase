\[\textbf{Доказать равенство: } \int_{0}^{+\infty} f(ax+\frac{b}{x}) \,dx =\frac{1}{a}\int_{0}^{+\infty} f(\sqrt{x^{2}+4ab}) \,dx. \]

Доказательство

\[ \textbf{Введём замену в левой части равенства: } x = \sqrt{\frac{b}{a}}e^{t}, t = \ln \sqrt{\frac{a}{b}}x: \space dx = \sqrt{\frac{b}{a}}e^{t}dt, \lim_{x \to 0^{+}}\ln \sqrt{\frac{a}{b}}x = -\infty, \space \lim_{x \to +\infty}\ln \sqrt{\frac{a}{b}}x = +\infty\] \[ \int_{0}^{+\infty} f(ax+\frac{b}{x}) \,dx = \sqrt{\frac{b}{a}}\int_{-\infty}^{+\infty}f(2\sqrt{ab}\, \text{ch} \space t)e^{t} \,dt = \sqrt{\frac{b}{a}}\left( \int_{-\infty}^{0}f(2\sqrt{ab} \,\text{ch} \space t)e^{t} \,dt + \int_{0}^{+\infty}f(2\sqrt{ab} \,\text{ch} \space t)e^{t} dt \right) =\] \[ = \sqrt{\frac{b}{a}}\left( \int_{0}^{+\infty}f(2\sqrt{ab} \,\text{ch} \space t)e^{-t} \,dt + \int_{0}^{+\infty}f(2\sqrt{ab} \,\text{ch} \space t)e^{t} \,dt \right) = 2\sqrt{\frac{b}{a}}\int_{0}^{+\infty}f(2\sqrt{ab} \,\text{ch} \space t) \,\text{ch} \space t \,dt.\] \[ \textbf{Введём замену в правой части равенства: }x = 2\sqrt{ab}\,\text{sh} t, \space t = \text{arsh}\frac{x}{2\sqrt{ab}}: dx = 2\sqrt{ab}\,\text{ch} \,t \,dt, \space \lim_{x \to 0^{+}} \text{arsh}\frac{x}{2\sqrt{ab}} = 0, \space \lim_{x \to +\infty}\text{arsh}\frac{x}{2\sqrt{ab}} = +\infty\] \[ \frac{1}{a}\int_{0}^{+\infty} f(\sqrt{x^{2}+4ab}) \,dx = \frac{2\sqrt{ab}}{a}\int_{0}^{+\infty} f(2\sqrt{ab} \,\text{ch} \space t) \,\text{ch} \space t \,dt = 2\sqrt{\frac{b}{a}}\int_{0}^{+\infty} f(2\sqrt{ab} \,\text{ch} \space t) \,\text{ch} \space t \,dt \textbf{, то есть выражения совпадают Q.E.D.}\]
