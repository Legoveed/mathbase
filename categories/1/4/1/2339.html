Задание

\[\textbf{Вычислить интеграл: } \int_{-\infty}^{+\infty} \frac{dx}{(x^{2}+x+1)^{2}}. \]

Решение

\[ \textbf{Выделим полный квадрат: } \int_{-\infty}^{+\infty} \frac{dx}{(x^{2}+x+1)^{2}} = \int_{-\infty}^{+\infty} \frac{dx}{\left(\left( x+\frac{1}{2} \right)^{2}+\frac{3}{4}\right)^{2}}. \] \[ \textbf{Введём замену } t = x+\frac{1}{2}: \space dt = dx, \space \lim_{x \to -\infty} \left(x+\frac{1}{2}\right) = -\infty, \space \lim_{x \to +\infty} \left(x+\frac{1}{2}\right) = +\infty\] \[ \int_{-\infty}^{+\infty} \frac{dt}{\left( t^{2}+\frac{3}{4}\right)^{2}}, \textbf{ введём замену } t = \frac{\sqrt{3}}{2} \mathit{tg} \space \theta, \space \theta = \mathit{arctg}\space \frac{2t}{\sqrt{3}}:\] \[\space dt = \frac{\sqrt{3}}{2}\sec^{2}\theta d\theta, \space t^{2}+\frac{3}{4} = \frac{3}{4}\mathit{tg}^{2}\theta+\frac{3}{4} = \frac{3}{4}\sec^{2}\theta, \lim_{t \to -\infty}\mathit{arctg}\space \frac{2t}{\sqrt{3}} = -\frac{\pi}{2}, \space \lim_{t \to +\infty}\mathit{arctg}\space \frac{2t}{\sqrt{3}} = \frac{\pi}{2};\] \[ \int_{-\frac{\pi}{2}}^{\frac{\pi}{2}} \frac{\frac{\sqrt{3}}{2}\sec^{2}\theta}{\left(\frac{3}{4}\sec^{2}\theta\right)^{2}}\,d\theta = \frac{8}{3\sqrt{3}}\int_{-\frac{\pi}{2}}^{\frac{\pi}{2}}\cos^{2}\theta \,d\theta = \frac{4}{3\sqrt{3}}\int_{-\frac{\pi}{2}}^{\frac{\pi}{2}}(1+\cos 2\theta) \,d\theta = \frac{4}{3\sqrt{3}}\left(\theta+\frac{\sin 2\theta}{2} \right)\Biggr|_{-\frac{\pi}{2}}^{\frac{\pi}{2}} = \frac{4}{3\sqrt{3}}\left(\frac{\pi}{2}+0-\left( -\frac{\pi}{2}+0 \right) \right) = \frac{4\pi}{3\sqrt{3}}.\]
