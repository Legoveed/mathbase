Задание

\[\textbf{Найти производные }y_{x}' ,\, y_{x^{2}}' ',\,y_{x^{3}}' ' ' \textbf{ от функции, заданной параметрически: } x = e^{t}\cos t ,\, y = e^{t} \sin t. \]

Решение

\[ x_{t}' = (e^{t}\cos t)' = (e^{t})'\cos t + e^{t}(\cos t)' = e^{t}(\cos t - \sin t),\, y_{t}' = (e^{t}\sin t)' = (e^{t})'\sin t + e^{t}(\sin t)' = e^{t}(\sin t + \cos t) \] \[ y_{x}' = \frac{dy}{dx} = \frac{\frac{dy}{dt}}{\frac{dx}{dt}} = \frac{e^{t}(\sin t + \cos t)}{e^{t}(\cos t - \sin t)} = \frac{\sin t + \cos t}{\cos t - \sin t}.\] \[ y_{x^{2}}' ' = \frac{d^{2}y}{dx^{2}} = \frac{\frac{d}{dt} \left( \frac{dy}{dx} \right)}{\frac{dx}{dt}} = \frac{\left( \frac{\sin t + \cos t}{\cos t - \sin t}\right)'}{e^{t}(\cos t - \sin t)};\] \[ \left( \frac{\sin t + \cos t}{\cos t - \sin t}\right)' = \frac{(\sin t + \cos t)'(\cos t - \sin t) - (\cos t - \sin t)'(\sin t + \cos t)}{(\cos t - \sin t)^{2}} = \frac{(\cos t - \sin t)^{2} + (\sin t + \cos t)^{2}}{(\cos t - \sin t)^{2}} = \] \[ =\frac{\cos^{2}t-2\cos t\sin t+\sin^{2}t + \sin^{2}t+2\sin t\cos t + \cos^{2} t}{(\cos t - \sin t)^{2}} = \frac{2}{(\cos t - \sin t)^{2}} \Rightarrow\] \[\Rightarrow y_{x^{2}}' ' = \frac{2e^{-t}}{(\cos t - \sin t)^{3}} = \frac{e^{-t}}{\sqrt{2}\cos^{3}\left(t+\frac{\pi}{4}\right)}.\] \[y_{x^{3}}' ' ' = \frac{d^{3}y}{dx^{3}} = \frac{\frac{d}{dt} \left( \frac{d^{2}y}{dx^{2}} \right)}{\frac{dx}{dt}} = \frac{\left( \frac{e^{-t}}{\sqrt{2}\cos^{3}\left(t+\frac{\pi}{4}\right)}\right)'}{e^{t}(\cos t - \sin t)};\] \[ \left( \frac{e^{-t}}{\sqrt{2}\cos^{3}\left(t+\frac{\pi}{4}\right)}\right)' = \frac{\left(e^{-t}\right)'\sqrt{2}\cos^{3}\left(t+\frac{\pi}{4}\right) - e^{-t}\left(\sqrt{2}\cos^{3}\left(t+\frac{\pi}{4}\right)\right)'}{2\cos^{6}\left(t+\frac{\pi}{4}\right)} = -\frac{e^{-t}\cos^{3}\left(t+\frac{\pi}{4}\right)-3e^{-t}\cos^{2}\left(t+\frac{\pi}{4}\right)\sin\left(t+\frac{\pi}{4}\right)}{\sqrt{2}\cos^{6}\left(t+\frac{\pi}{4}\right)} =\] \[ = -\frac{e^{-t}\cos\left(t+\frac{\pi}{4}\right)-3e^{-t}\sin\left(t+\frac{\pi}{4}\right)}{\sqrt{2}\cos^{4}\left(t+\frac{\pi}{4}\right)} = \frac{e^{-t}\left(\frac{3\sqrt{2}}{2}(\cos x + \sin x) - \frac{\sqrt{2}}{2}(\cos x - \sin x) \right)}{\sqrt{2}\cos^{4}\left(t+\frac{\pi}{4}\right)} = \frac{e^{-t}\left(2\sin x + \cos x \right)}{\cos^{4}\left(t+\frac{\pi}{4}\right)} \Rightarrow\] \[\Rightarrow y_{x^{3}}' ' ' = \frac{e^{-2t}\left(2\sin x + \cos x \right)}{\sqrt{2}\cos^{5}\left(t+\frac{\pi}{4}\right)} \]
